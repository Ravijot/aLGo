#include<iostream>
#include<conio.h>
#include<algorithm>
using namespace std;
class node
{
	public:
	int data;
	node *left;
	node *right;
		node(int d)
		{
			data=d;
			right=NULL;
			left=NULL;
		}
};
node* buildtree()
{
	int d;
	cin>>d;
	if(d==-1)
	{
		return NULL;
		
	}
	node *root=new node(d);
	root->data=d;
	root->left=buildtree();
	root->right=buildtree();
	return root;
	
	
}
void print(node *root)
{
	if(root==NULL)
	{
		return;
	}
	cout<<root->data<<" ";
	print(root->left);
	print(root->right);
	
}
int height(node *root)
{
	if(root==NULL)
	{
		return 0;
	}
	int lsth=height(root->left);
	int rsth=height(root->right);
	return max(lsth,rsth)+1;
}
void printkthlevel(node *root,int k)
{
	if(root==NULL)
	{
		return;
	}
	if(k==1)
	{
		cout<<root->data<<" ";
		return ;
	}
	printkthlevel(root->left,k-1);
    printkthlevel(root->right,k-1);
    return;
}
void printALLlevels(node *root)
{
	int H=height(root);
	for(int i=0;i<=H;i++)
	{
		printkthlevel(root,i);
		cout<<endl;
	}
}

{
	node *root=buildtree();
	print(root);

	printALLlevels(root);
