node * insertsorted(struct node *head,int data)
{
	node *temp = new node();
	temp->data=data;
	temp->next=NULL;
	if(head == NULL)
                {
                      head = temp;
                      return temp;
                 }
                if(data <= head->data)
                {
                      temp->next = head;
                      head = temp;
                      return head;
                 }
                node *curr = head->next, *prev = head;
                while(curr != NULL)
                {
                     if(curr->data > data)
                     {
                            temp->next = curr;
                            prev->next = temp;
                            return head;
                     }
                     prev = curr;
                     curr = curr->next;
    }
    prev->next = temp;
    return head;
}
