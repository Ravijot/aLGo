#include<iostream>

using namespace std;
class list
{
     private:
	struct node 
	{
	     int data;
	     struct node *next;
	}*head;
	public:
	list()
	{
	     head=NULL;
	}
	void insert();
	void display();
	void reverse();
};
void list::reverse()
{
	struct node *pre,*cur,*next;
	pre=NULL;
	cur=head;
	while(cur!=NULL)
	{
		next=cur->next;
		cur->next=pre;
		pre=cur;
		cur=next;
	}
	head=pre;
}
void list::insert()
{
	int n;
	cin>>n;
	struct node *temp,*last;
	temp=new struct node();

	if(head==NULL)
	{
		temp->data=n;
		temp->next=NULL;
		head=temp;
	    last=head;
	}
	else
	{
		temp->data=n;
		temp->next=NULL;
		last->next=temp;
		last=temp;
	}
}
void list::display()
{
	struct node *t;
	t=new struct node();
	t=head;
	if(t==NULL)
	{
		cout<<"List is empty";
	}
	else
	{
		while(t!=NULL)
		{
			cout<<t->data<<" ";
			t=t->next;
		}
	}
}
