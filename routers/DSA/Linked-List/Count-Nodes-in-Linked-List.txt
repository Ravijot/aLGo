#include<iostream>
#include<conio.h>
using namespace std;
class list
{
     private:
	struct node
	{
		int data;
		struct node *next;
	}*head;
     public:
	list()
	{
	     head=NULL;
	}
	void insert(int n);
	void display();
	int count();
};
void list::insert(int n)
{
	struct node *temp;
	temp=new struct node();
	temp->data=n;
	temp->next=head;
	head=temp;
}
void list::display()
{
	if(head==NULL)
	{
	     cout<<"List is empty";
	}
	else
	{
	     struct node *m;
	     m=head;
	     cout<<"\nLinked List ";
	     while(m!=NULL)
	     {
		cout<<m->data<<" ";
		m=m->next;
	     }
	 }
}
int list::count()
{
	struct node *p;
	p=head;
	int n=0;
	while(p!=NULL)
	{
		n++;
		p=p->next;
	}
	return n;
}
int main()
{
	list l;
	int n,a;
	cout<<"Enter the number of elements you want";
	cin>>n;
	for(int i=0;i<n;i++)
	{
	     cin>>a;
                     l.insert(a);
	}
	l.display();
	cout<<"\nNumber of nodes are "<<l.count();
	getch();
	return 0;
}
