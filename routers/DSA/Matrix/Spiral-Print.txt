#include<iostream>

using namespace std;
//Spiral Print Matrix
void spiral_print(int matrix[20][20],int c,int r)
{
	cout<<"Spiral Print - ";
	int startrow=0,endrow=r-1,startcol=0,endcol=c-1;
	while(startrow<=endrow && startcol<=endcol)
	{
		// Print start row first
		
		for(int i=startcol;i<=endcol;i++)
		{
			cout<<matrix[startrow][i]<<" ";
		}
		startrow++;
		
		//Print end column
		for(int i=startrow;i<=endrow;i++)
		{
			cout<<matrix[i][endcol]<<" ";
				}		
		endcol--;
	
		// Print end row
		if(endrow>startrow)
		{
		for(int i=endcol;i>=startcol;i--)
		{
			cout<<matrix[endrow][i]<<" ";
		}
		endrow--;
	}
		//Print first column
	if(startcol<endcol)
	{
	
		for(int i=endrow;i>=startrow;i--)
		{
			cout<<matrix[i][startcol]<<" ";
		}
		startcol++;
}
		
	}
}
int main()
{
	int a[20][20];
	cout<<"Enter the number of columns";
	int n,m;
	cin>>m;
	cout<<"Enter the number of rows";
	cin>>n;
	cout<<"Enter the elements of the array";
	for(int i=0;i<m;i++)
	{
		for(int j=0;j<n;j++)
		{
			cin>>a[i][j];
			
		}
	}
	cout<<"\nMatrix is\n";
	for(int i=0;i<m;i++)
	{
		for(int j=0;j<n;j++)
		{
			cout<<a[i][j]<<"\t";
			
		}
		cout<<endl;
	}
	
	spiral_print(a,m,n);
	getch();
	return 0;
}
