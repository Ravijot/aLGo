#include<iostream>
#include<conio.h>
using namespace std;
int binarysearch(int a[],int beg,int end,int item)
{
	if(end>=beg)
	{
		int mid = (end + beg)/2;
		if(a[mid]==item)
	    {
		   return mid;
	    }
	    else if(a[mid]<item)
	    {
	    	return binarysearch(a,beg,end-1,item);
	    }
	    else 
        {
		   return binarysearch(a,mid+1,end,item);
	    }
	}
	return -1;
}
int main()
{
   int a[50];
   cout<<"Enter the number of elements you want to enter";
   int n;
   cin>>n;
   cout<<"\nEnter the elements of the array";
   for(int i=0;i<n;i++)
   {
	cin>>a[i];
    }
    cout<<"\nEnter the item to be search";
    int item;
    cin>>item;
    int beg=0,end=n-1;
    int result=binarysearch(a,beg,end,item);
    if(result==-1)
    {
	cout<<"Item is not present in the array";
     }
    else
     {
	cout<<"Item is present at index "<<result;
      }
      getch();
      return 0;
}
