/*  Program to calculate power of a number using recursion
        we  can write a^b= a*a^(b-1)
        Recursive Function f(a,b)= a.f(a,b-1)
        */
#include<iostream>
#include<conio.h>
using namespace std;
int pow(int m,int n)
{
	if(n==0)
	{
		
		return 1;
	}
	else
	{
		return m*pow(m,n-1);
	}
}
int main()
{
	int a,b;
	cin>>a>>b;
	cout<<a<<"^"<<b<<" is "<<pow(a,b);
	getch();
	return 0;	
}
