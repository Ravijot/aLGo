#include<iostream>
#include<algorithm>
using namespace std;
void targetsum(int a[],int num,int target)
{
	for(int i=0;i<num;i++)
	{
		for(int j=i+1;j<num;j++)
		{
			if(a[i]+a[j]==target)
			{
				cout<<a[i]<<" and "<<a[j]<<endl;
			}
		}
	}
}
int main()
{
	int n;
	cout<<"Enter number of elements of array";
	cin>>n;
	int a[n];
	int target;
	cout<<"Enter the Elements of Array\n";
	for(int i=0;i<n;i++)
	{
		cin>>a[i];
	}
	cout<<"Enter the target sum ";
	cin>>target;
	sort(a,a+n);
	targetsum(a,n,target);
	return 0;
}
