#include<iostream>
#include<conio.h>
using namespace std;
int main()
{
	int a[20];
	cout<<"Enter the number of elements of the array";
	int n;
	cin>>n;
	for(int i=0;i<n;i++)
	{
		cin>>a[i];
	}
	int min=a[0];
	int max=a[0];
	for(int i=1;i<n;i++)
	{
		if(a[i]<min)
		{
			min=a[i];
		}
		else if(a[i]>max)
		{
			max=a[i];
		}
	}
	cout<<"MAX = "<<max<<endl;
	cout<<"MIN = "<<min<<endl;
	getch();
	return 0;
}
