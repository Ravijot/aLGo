#include<iostream>
#include<conio.h>
using namespace std;
int main()
{
	cout<<"Enter the number of elements of the array";
	int n;
	cin>>n;
	int a[50];
	cout<<"Enter the elements of array\n";
	for(int i=0;i<n;i++)
	{
		cin>>a[i];
	}
	int k=0,j=n-1,temp=0;
	while(k<j)
	{
		temp=a[k];
		a[k]=a[j];
		a[j]=temp;
		k++;
		j--;
		
	}
	cout<<"Elements after reverse of array\n";
	for(int i=0;i<n;i++)
	{
		cout<<a[i]<<" ";
	}
	getch();
	return 0;
}
