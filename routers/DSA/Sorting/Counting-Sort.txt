#include<iostream>
using namespace std;
void countsort(int *a,int n)
{
	//largest in the array
	int largest=a[0]; 
    //We are only considering the positive numbers
	for(int i=1;i<n;i++)
	{
		if(a[i]>largest)
		{
			largest=a[i];
		}
	}
	int freq[largest+1]={0};
	for(int i=0;i<n;i++)
	{
		freq[a[i]]++;
	}
	int j=0;
	for(int i=0;i<=largest;i++)
	{
		while(freq[i]>0)
		{
			a[j]=i;
			freq[i]--;
			j++;
		}
	}
}
int main()
{
	int a[]={0,2,1,0,2,3,4,5,3,5,4};
	int n=sizeof(a)/sizeof(int);
	countsort(a,n);
	cout<<"Sorted Array : ";
	for(int i=0;i<n;i++)
	{
		cout<<a[i]<<" ";
	}
	return 0;
}
