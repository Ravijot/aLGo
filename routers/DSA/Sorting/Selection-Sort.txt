/*
Selection Sort
Number of Comparisons 1+2+3+4+.....+n= n(n-1)/2 =O(n^2)
Number of Swap  =n-1 = O (n)
This is the only sorting algorithm with 
minimim number of swaps  
*/

#include<iostream>
using namespace std;
void selectionsort(int a[],int n)
{
   for(int i=0;i<n-1;i++)
   {
	//find out the smallest element in the array
	for(int j=i;j<n-1;j++)
	{
		int min_index=i;
		for(int j=i;j<=n-1;j++)
		{
			if(a[j]<a[min_index])
			{
				min_index=j;
			}
		}
		//Swap after above root
		swap(a[i],a[min_index]);
	}
		
    }
}
int main()
{
    int n;
    cout<<"Enter the number of elements\n";
	cin>>n;
	int a[n];
	cout<<"Enter the elements\n";
	for(int i=0;i<n;i++)
	{
		cin>>a[i];
		}	
	selectionsort(a,n);	
	cout<<"Sorted Array - ";
	for(int i=0;i<n;i++)
	{
		cout<<a[i]<<" ";
	}
	return 0;
}
