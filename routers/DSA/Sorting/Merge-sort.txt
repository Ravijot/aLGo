#include<iostream>
using namespace std;
void merge(int *a,int s,int e)
{
	int mid=(s+e)/2;
	int i=s;
	int j=mid+1;
	int k=s;
	int temp[100];
	while(i<=mid && j<=e)
	{
		if(a[i]< a[j])
		{
			temp[k++]=a[i++];
			
		}
		else
		{
			temp[k++]=a[j++];
		}
	}
	while(i<=mid)
	{
		temp[k++]=a[i++];
	}
	while(j<=e)
	{
		temp[k++]=a[j++];
	}
	//Copy all elements to the 
	for(int i=s;i<=e;i++)
	{
		a[i]=temp[i];
	}
}
void mergesort(int *a,int s,int e)
{
	//Base Case
	if(s>=e)
	{
		return;
	}
	//Divide
	int mid=(s+e)/2;
                //Recursively Sort the two Array 
                //start to mid, mid+1 to end
	mergesort(a,s,mid);
	mergesort(a,mid+1,e);
	//Merge the Two sorted array
	merge(a,s,e);
	
}
int main()
{
   cout<<"Enter the number of elements you want to enter\n";
   int n;
   cin>>n;
   cout<<"Enter the numbers";
   int a[n];
   for(int i=0;i<n;i++)
   {
	cin>>a[i];
   }
   mergesort(a,0,n-1);
   cout<<"After Mergesort - ";
   for(int i=0;i<n;i++)
   {
	cout<<a[i]<<" ";
   }	
   return 0;
}
