#include<iostream>
#include<stack>
using namespace std;
int main()
{
	int n;
	cin>>n;
	int a[n];
	stack<int> s;
	for(int i=0;i<n;i++)
	{
		cin>>a[i];
	}
	
	int arr[n];
	for (int i = n - 1; i >= 0; i--)  
    { 
        
        while (!s.empty() && s.top() <= a[i]) 
        {
           	s.pop(); 
		}
        if (s.empty())  
        {
        	arr[i] = -1; 
		 }         
        else 
        {
        	arr[i] = s.top(); 
		}        
  
        s.push(a[i]); 
    } 
    for(int i=0;i<n;i++)
    {
    	cout<<arr[i]<<" ";
	}
	return 0;
}
/*
Sample Input
3
1 2 3
Sample Output
2 3 -1
Explanation
Next greater element for 1 is 2, 
for 2 is 3 but not present for 3 therefore -1*/
