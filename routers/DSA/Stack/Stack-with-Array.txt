#include<iostream>
#define MAX 100
using namespace std;
class stack
{
	int TOP;
	int a[MAX];
	public:
		stack()
		{
		TOP=-1;	
		}
		void push(int n);
		void pop();
		void display();
};
void stack::push(int n)
{
	if(MAX==TOP)
	{
		cout<<"OVERFLOW";
	}
	else
	{
		TOP++;
		a[TOP]=n;
	}
}
void stack::pop()
{
	if(TOP==-1)
	
	{
		cout<<"UNDERFLOW";
		
	}
	else
	{
		TOP--;
		cout<<"Element Poped Out Successfully";
	}
}
void stack::display()
{
	cout<<"Elements of the Stack are : ";
	for(int i=0;i<TOP;i++)
	{
		cout<<a[i]<<" ";
	}
}

{
	stack s;
	int choice,n;
	label:
	system("CLS");
	cout<<"Enter 1 for Push Operation";
	cout<<"\nEnter 2 for Pop Operation";
	cout<<"\nEnter 3 for Display Operation";
	cin>>choice;
	switch(choice)
	{
	        case 1:
		cout<<"\nEnter the number you want to enter";
		cin>>n;
		s.push(n);
		goto label;
		case 2:
		 s.pop();
		 goto label;
		 case 3:
		  s.display();
		  cout<<"\nEnter 1 for continue and 0 for exit";
	                   int b;
		   cin>>b;
		   if(b==1)
		   {
		      goto label;
                	    }
	                    else
		     {

		     }

	}
	return 0;
}
