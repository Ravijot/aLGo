#include<iostream>
#include<cstring>
using namespace std;
bool ispallindrome( char a[])
{
	int i=0;
	int j = strlen(a) - 1;
	while(i<j)
	{
		if(a[i]==a[j])
		{
			i++;
			j--;
		}
		else
		{
			return false;
		}
		return true;
	}
}
int main()
{
	char a[100];
	cin.getline(a,100);
	if(ispallindrome(a))
	{
		cout<<"String is Pallindrome";
	}
	else
	{
		cout<<"String is not pallindrome";
	}
	return 0;
}
