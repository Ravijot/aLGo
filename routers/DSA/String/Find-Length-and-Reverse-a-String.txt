#include<iostream>
#include<conio.h>
using namespace std;
int main()
{
	cout<<"Enter the the string\n";
	char a[50];
	cin>>a;
	int length=0;
	for(int i=0;a[i]!='\0';i++)
	{
		length++;
	}
	cout<<"length = "<<length;
	int i=0,j=length-1;
	char temp;
	while(i<j)
	{
		temp=a[i];
		a[i]=a[j];
		a[j]=temp;
		i++;
		j--;
	}
	cout<<"\nReverse string : ";
	for(int i=0;i<length;i++)
	{
	  cout<<a[i];
	}
	getch();
	return 0;
}
