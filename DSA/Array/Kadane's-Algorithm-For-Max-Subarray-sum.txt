#include<iostream>
#include<conio.h>
using namespace std;

int main()
{ 
     cout<<"Enter the number of elements of Array";
     int n;
     cin>>n;
     int a[1000],cursum=0,maxsum=0;
     cout<<"Enter the Elements of Array\n";
     for(int i=0;i<n;i++)
     {
     	cin>>a[i];
	 }
	 //Kadane's Algorithm
	 for(int i=0;i<n;i++)
	 {
	 	cursum=cursum+a[i];
		  if(cursum<0)
		  {
		  	cursum=0;
		  }
		  else
		  {
		  	if(maxsum<cursum)
		  	{
		  		maxsum=cursum;
			  }
		  }
	}
	cout<<"\nMaximum Subarray sum :"<<maxsum;
	return 0;
}
