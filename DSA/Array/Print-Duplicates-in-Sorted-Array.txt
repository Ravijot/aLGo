#include<iostream>
#include<conio.h>
using namespace std;
int main()
{
	int a[30];
	int n,j;
	cout<<"Enter the number of elements of array";
	cin>>n;
	cout<<"Enter the elements of array\n";
	for(int i=0;i<n;i++)
	{
		cin>>a[i];
	}
	int count=0;
	for(int i=0;i<n-1;i++)
	{
	count=0;
	if(a[i]==a[i+1])
	{
	j=i+1;
	while(a[j]==a[i])
	{
	count++;
	j++;
	}	
	cout<<endl<<a[i]<<" appearing "<<count+1<<" times";
	i=j-1;
	}
	}
	getch();
	return 0;
}
