#include<iostream>
#include<conio.h>
using namespace std;
void display(int a,int b,int c)
{
	cout<<"Element "<<c<<" present at location "<<a;
}
int main()
{
	int a[20];
	cout<<"Enter the number of elements you want to enter";
	int n;
	cin>>n;
	cout<<"Enter the elements of array";
	for(int i=0;i<n;i++)
	{
		cin>>a[i];
	}
	cout<<"Elements of array are";
	for(int i=0;i<n;i++)
	{
		cout<<a[i]<<"\t";
	}
	cout<<"\nEnter the element you want to search";
	int item;
	cin>>item;
	int pos=0,loc=0;
	for(int i=0;i<n;i++)
	{
	if(item==a[i])
	{
		pos++;
		loc=i+1;
		display(loc,pos,item);
		}	
	}
	if(pos==0)
	{
		cout<<"Element is not present in the array";
	}
	getch();
	return 0;
}
