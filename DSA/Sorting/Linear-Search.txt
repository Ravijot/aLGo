#include<iostream>
using namespace std;
int main()
{
   int n;
   cout<<"Enter the number";
   cin>>n;
   int a[n];
   for(int i=0;i<n;i++)
   {
	cin>>a[i];
   }
   int key;
   cout<<"\nEnter the number you want to search";
   cin>>key;
   int i;
   for( i=0;i<n;i++)
   {
	if(a[i]==key)
	{
	   cout<<key<<" is present at index "<<i+1;
	   break;
	}
    }
    if(i==n)
    {
	cout<<key<<" is present in array ";
    }
    return 0;
}
