/* Bubble Sort
Number of comparisons = 1+2+3+4+....+n=n(n-1)/2=O(n^2)
Number of swaps=1+2+3+4+....+n=n(n-1)/2=O(n^2)
K passes give K largest elements*/ 
#include<iostream>
using namespace std;
void bubblesort(int a[],int n)
{
	for(int i=1;i<=n-1;i++)
	{
		for(int j=0;j<=(n-i-1);j++)
		{
			if(a[j]>a[j+1])
			{
				swap(a[j],a[j+1]);
			}
		}
	}
}
int main()
{
	int n;
	cout<<"Enter the number of elements";
	cin>>n;
	int a[n];
	cout<<"Enter the elements of array";
	for(int i=0;i<n;i++)
	{
		cin>>a[i];
	}
	bubblesort(a,n);
	cout<<"Sorted Array : ";
	for(int i=0;i<n;i++)
	{
		cout<<a[i]<<" ";
	}
	return 0;
}
