#include<iostream>
#include<conio.h>
using namespace std;
int main()
{
	int value=1;
	int a[3][3];
	for(int i=0;i<3;i++)
	{
		for(int j=0;j<3;j++)
		{
			a[i][j]=value++;
		}
	}
	cout<<"Matrix\n";
	for(int i=0;i<3;i++)
	{
		for(int j=0;j<3;j++)
		{
			cout<<a[i][j]<<" ";
		}
		cout<<endl;
	}
	int temp=0; //Taking a variable to store the value
	//Transpose of matrix
	for(int i=0;i<3;i++)
	{
		for(int j=0;j<i;j++)
		{
			temp=a[i][j];
			a[i][j]=a[j][i];
			a[j][i]=temp;
		}
	}
	cout<<"Transpose of Matrix\n";
    	for(int i=0;i<3;i++)
	{
		for(int j=0;j<3;j++)
		{
			cout<<a[i][j]<<" ";
		}
		cout<<endl;
	}
    
   	getch();
	return 0;
}
