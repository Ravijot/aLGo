#include<iostream>
#include<conio.h>
using namespace std;
int main()
{
	int value=1;
	int a[3][3];
	for(int i=0;i<3;i++)
	{
		for(int j=0;j<3;j++)
		{
			a[i][j]=value++;
		}
	}
	cout<<"\nMatrix is\n";
	for(int i=0;i<3;i++)
	{
		for(int j=0;j<3;j++)
		{
			cout<<a[i][j]<<"\t";
			
		}
		cout<<endl;
	}
	cout<<"\nColumn wise priting\n";
	for(int i=0;i<3;i++)
	{
		for(int j=0;j<3;j++)
		{
			cout<<a[j][i]<<"\t";
		}
		cout<<endl;
	}
	getch();
	return 0;
}
