#include<iostream>
#include<conio.h>
using namespace std;
int main()
{
	int a[20][20];
	cout<<"Enter the number of columns";
	int n,m;
	cin>>m;
	cout<<"Enter the number of rows";
	cin>>n;
	cout<<"Enter the elements of the array";
	for(int i=0;i<m;i++)
	{
		for(int j=0;j<n;j++)
		{
			cin>>a[i][j];
			
		}
	}
	cout<<"Matrix is\n";
	for(int i=0;i<m;i++)
	{
		for(int j=0;j<n;j++)
		{
			cout<<a[i][j]<<" ";
			
		}
		cout<<endl;
	}
	cout<<"Wave Printing \n";  //Wave Print
	for(int j=0;j<n;j++)
	{
		if(j&1)//Odd column
		{
			for(int i=m-1;i>=0;i--)
			{
				cout<<a[i][j]<<" ";
			}
		}
		else//Even column
		{
			for(int i=0;i<m;i++)
			{
			cout<<a[i][j]<<" ";	
			}
		}
	}
	getch();
	return 0;
}
