#include<iostream>
#include<conio.h>
using namespace std;
class node
{
	public:
	int data;
	node *left;
	node *right;
		node(int d)
		{
			data=d;
			right=NULL;
			left=NULL;
		}
};
node* buildtree()
{
	int d;
	cin>>d;
	if(d==-1)
	{
		return NULL;
		
	}
	node *root=new node(d);
	root->data=d;
	root->left=buildtree();
	root->right=buildtree();
	return root;
	
	
}
void print(node *root)
{
	if(root==NULL)
	{
		return;
	}
	cout<<root->data<<" ";
	print(root->left);
	print(root->right);
	
}
int sum(node *root)
{
	if(root==NULL)
	{
		return 0;
	}
	return root->data+sum(root->left)+sum(root->right);
}
int main()
{
	node *root=buildtree();
	print(root);
	
	cout<<"\nSum of the nodes - "<<sum(root);
	return 0;
}
