#include<iostream>
#include<conio.h>
#include<string.h>
using namespace std;
class node
{
	public:
	int data;
	node *left;
	node *right;
		node(int d)
		{
			data=d;
			right=NULL;
			left=NULL;
		}
};
node* buildtree()
{
int d;
cin>>d;
if(d==-1)
{
	return NULL;
}
node *root=new node(d);
root->left=buildtree();
root->right=buildtree();
return root;
}
void print(node *root)
{
	if(root==NULL)
	{
		return;
	}
	cout<<root->data<<" ";
	print(root->left);
	print(root->right);
	
}

int main()
{
	node *root=buildtree();
	print(root);
	return 0;
}
