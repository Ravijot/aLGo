#include<iostream>
#include<conio.h>
#include<algorithm>
using namespace std;
class node
{
	public:
	int data;
	node *left;
	node *right;
		node(int d)
		{
			data=d;
			right=NULL;
			left=NULL;
		}
};
node* buildtree()
{
	int d;
	cin>>d;
	if(d==-1)
	{
		return NULL;
		
	}
	node *root=new node(d);
	root->data=d;
	root->left=buildtree();
	root->right=buildtree();
	return root;
	
	
}
void print(node *root)
{
	if(root==NULL)
	{
		return;
	}
	cout<<root->data<<" ";
	print(root->left);
	print(root->right);
	
}
int height(node *root)
{
	if(root==NULL)
	{
		return 0;
	}
	int lsth=height(root->left);
	int rsth=height(root->right);
	return max(lsth,rsth)+1;
}
int diameter(node *root)
{
	if(root==NULL)
	{
		return 0;
	}
	int h1=height(root->left);
	int h2=height(root->right);
	int op1=h1+h2;
	int op2=diameter(root->left);
	int op3=diameter(root->right);
	return max(op1,max(op2,op3));
}
int main()
{
	node *root=buildtree();
	print(root);
	cout<<"\nDiameter of Tree - "<<diameter(root);
	return 0;
}
