#include<iostream>
#include<stack>
#include<string>
using namespace std;
bool checkexpression(string a)
{
	stack<char> s;
	char cur;
	for(int i=0;i<a.size();i++)
	{
		cur=a[i];
		if(cur=='(' || cur=='[' || cur=='{')
		{
			s.push(cur);
		}
		else if(cur==')')
		{
			if(s.top()=='(')
			{
				s.pop();
			}
			else
			{
				s.push(cur);
			}
			continue;
		}
		else if(cur==']')
		{
			if(s.top()=='[')
			{
				s.pop();
			}
			else
			{
				s.push(cur);
			}
		
			continue;
		}
		else if(cur=='}')
		{
			if(s.top()=='{')
			{
				s.pop();
			}
			else
			{
				s.push(cur);
			}
			continue;
		}
		
	}
	if(s.empty())
	{
		return true;
	}
	else
	{
		return false;
	}
}
int main()
{
	string expr;
	cin>>expr;
	 bool isvalid=checkexpression(expr);
	if(isvalid)
	{
		cout<<"YES";
	}
	else
	{
		cout<<"NO";
	}
	return 0;
}
