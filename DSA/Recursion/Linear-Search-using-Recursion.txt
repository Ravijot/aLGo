#include<iostream>
#include<conio.h>
using namespace std;
int linearsearch(int a[],int n,int item,int i)
{
	if(i==n)
	{
		return -1;
	}
	else if(a[i]==item)
	{
		return i;
	}
	else
	{
		return linearsearch(a,n,item,i+1);
	}
}
int main()
{
   int a[50];
   cout<<"Enter the number of elements you want to enter";
   int n;
   cin>>n;
   cout<<"\nEnter the elements of the array";
   for(int i=0;i<n;i++)
   {
	cin>>a[i];
    }
   cout<<"\nEnter the item to be search";
   int item;
   cin>>item;
   int result=linearsearch(a,n,item,0);
   if(result==-1)
   {
	cout<<"\nItem is not present in the array";
    }
   else
   {
	cout<<"\nItem is present at index "<<result;
    }
    getch();
    return 0;
}
