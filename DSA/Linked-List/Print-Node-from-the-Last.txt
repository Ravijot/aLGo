class list
{
    private:
	struct node 
	{
	     int data;
	     struct node *next;
	}*head;
     public:
	list()
	{
	    head=NULL;
	}
	void insert();
	void display();
	void printnode();
};
void list::printnode()
{
	int k;
	cout<<"\nEnter the K node which you want to print from end";
	cin>>k;
	struct node *slow,*fast;
	slow=fast=head;
	for(int i=0;i<k;i++)
	{
		fast=fast->next;
	}
	while(fast!=NULL)
	{
		slow=slow->next;
		fast=fast->next;
	}
	cout<<"\n"<<k<<" Node from last of linked list "<<slow->data;
}
void list::insert()
{
	int n;
	cin>>n;
	struct node *temp,*last;
	temp=new struct node();

	if(head==NULL)
	{
		temp->data=n;
		temp->next=NULL;
		head=temp;
	    last=head;
	}
	else
	{
		temp->data=n;
		temp->next=NULL;
		last->next=temp;
		last=temp;
	}
}
void list::display()
{
	struct node *t;
	t=new struct node();
	t=head;
	if(t==NULL)
	{
		cout<<"List is empty";
	}
	else
	{
		while(t!=NULL)
		{
			cout<<t->data<<" ";
			t=t->next;
		}
	}
}
