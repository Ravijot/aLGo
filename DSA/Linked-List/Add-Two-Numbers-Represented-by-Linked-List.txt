struct node
{
	int data;
	struct node *next;
};
struct node* reverse(struct node* head)
{
	struct node *pre,*cur,*next;
	pre=NULL;
	cur=head;
	while(cur!=NULL)
	{
		next=cur->next;
		cur->next=pre;
		pre=cur;
		cur=next;
	}
	head=pre;
	return head;
}
void display(struct node *head)
{
	if(head==NULL)
	{
		cout<<"List is empty";
	}
	else
	{
		struct node *t;
		t=head;
		cout<<"Linked List after addition - ";
		while(t!=NULL)
		{
			cout<<t->data<<" ";
			t=t->next;
		}
	}
}
