#include<iostream>
#include<conio.h>
using namespace std;
class list
{
     private:
	struct node 
	{
		int data;
		struct node *next;
	}*head;
     public:
	list()
	{
		head=NULL;
	}
	void insert();
	void display();
	void mid();
};
void list::mid()
{
	struct node *slow,*fast;
	slow=head;
	fast=head;
	while(fast->next!=NULL)
	{
		slow=slow->next;
		fast=fast->next;
		fast=fast->next;
	}
	cout<<"\nMid of linked list "<<slow->data;
}
void list::insert()
{
	int n;
	cin>>n;
	struct node *temp,*last;
	temp=new struct node();
	if(head==NULL)
	{
		temp->data=n;
		temp->next=NULL;
		head=temp;
		last=head;
	}
	else
	{
		temp->data=n;
		temp->next=NULL;
		last->next=temp;
		last=temp;
	}
}
void list::display()
{
	struct node *t;
	t=new struct node();
	t=head;
	if(t==NULL)
	{
		cout<<"List is empty";
	}
	else
	{
		while(t!=NULL)
		{
			cout<<t->data<<" ";
			t=t->next;
		}
	}
}
