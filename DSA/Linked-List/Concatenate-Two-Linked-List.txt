#include<iostream>
#include<conio.h>
using namespace std;
class list
{
	private:
		struct node 
		{
			int data;
			struct node *next;
		}*head;
	public:
		list()
		{
			head=NULL;	
		}
		void insert();
		void display();
		list operator+(list l);
};
list list::operator+(list l)
{
	list temp;
	struct node *p;
	p=head;
	while(p->next!=NULL)
	{
		p=p->next;
	}
	p->next=l.head;
	l.head=NULL;
	temp.head=head;
	return temp;
}
void list::insert()
{
	int n;
	cin>>n;
	struct node *temp,*last;
	temp=new struct node();
	if(head==NULL)
	{
		temp->data=n;
		temp->next=NULL;
		head=temp;
	    last=head;
	}
	else
	{
		temp->data=n;
		temp->next=NULL;
		last->next=temp;
		last=temp;
	}
}
void list::display()
{
	struct node *t;
	t=new struct node();
	t=head;
	if(t==NULL)
	{
		cout<<"List is empty";
	}
	else
	{
		while(t!=NULL)
		{
			cout<<t->data<<" ";
			t=t->next;
		}
	}
}
int main()
{
     list l1,l2,l3;

	l2.insert();
     }
     l2.display();
     l3=l1+l2;

     l3.display();

}
