node* reverse(struct node *head)
{
	struct node *pre,*cur,*next;
	pre=NULL;
	cur=head;
	while(cur!=NULL)
	{
		next=cur->next;
		cur->next=pre;
		pre=cur;
		cur=next;
	}
	head=pre;
	return head;
}
