using namespace std;
class list
{
   private:
	struct node
	{
		int data;
		struct node *next;
	}*head=NULL,*last=NULL;
    public:
	list()
	{
		head=NULL;
	}		
	void insertpos();
	void display();
	void insert();
	void insertend();
	void insertbeg();
};
void list::insert()
{
	cout<<"Enter the number";
	int n;
	cin>>n;
	struct node *temp;
	temp=new struct node();
	temp->data=n;
	temp->next=NULL;
	if(head==NULL)
	{
	     head = last = temp;
	}
	else
	{
	    last->next = temp;
	    last = temp;
	}
}
void list::insertend()
{
	struct node *x,*temp;
	x=new struct node();
	x=head;
	cout<<"Enter the number you want to enter ";
	int n;
	cin>>n;
	temp=new struct node();
	temp->data=n;
	temp->next=NULL;
	while(x->next!=NULL)
	{
		x=x->next;
	}
	x->next=temp;
	
}
void list::insertbeg()
{
	struct node *temp;
	temp=new struct node();
	cout<<"Enter the number which you want to enter";
	int n;
	cin>>n;
	temp->data=n;
	temp->next=head;
	head=temp;
}
void list::insertpos()
{
    struct node *temp,*p;
    temp=new struct node();
    p=new struct node();
    p=head;
    cout<<"Enter the number which you want to enter";
    int n;
    cin>>n;
    temp->data=n;
    temp->next=NULL;
    cout<<"Enter the position where you want to enter node ";
    int pos;
    cin>>pos;
    for(int i=1;i<pos-1;i++)
    {
	p=p->next;
    }
    temp->next=p->next;
    p->next=temp;
}

void list::display()
{
	if(head==NULL)
	{
	    cout<<"List is empty";
	}
	else
	{
	    struct node *m;
	    m=head;
	    cout<<"Linked List:";
	    while(m!=NULL)
	    {
		cout<<m->data<<" ";
		m=m->next;
	     }
	}
};
