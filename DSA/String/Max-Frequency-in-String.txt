#include<iostream>
#include<algorithm>
using namespace std;
int main()
{
	char a[1000];
	cin.getline(a,1000);
	int length=0;
	int maxsum=0,cursum=0;
	char max;
	for(int i=0;a[i]!='\0';i++)
	{
		cursum=1;
		for(int j=1;a[j]!='\0';j++)
		{
			
			if(a[i]==a[j])
			{
				cursum++;
			}
		}
		if(cursum>maxsum)
		{
			maxsum=cursum;
			max=a[i];
		}
	}
	cout<<max;
	return 0;	
	
}
